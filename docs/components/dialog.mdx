## Usage

  <Dialog>
    <DialogTrigger asChild>
      <Win95Button>Open Dialog</Win95Button>
    </DialogTrigger>
    <DialogContent>
      <DialogHeader>
        <DialogTitle>Save As</DialogTitle>
        <DialogDescription>
          Choose a location to save your file.
        </DialogDescription>
      </DialogHeader>
      <div className="space-y-2">
        <label className="text-xs text-win95-black">File name:</label>
        <Win95Input defaultValue="Untitled.txt" />
      </div>
      <div className="flex justify-end gap-2 mt-4">
        <Win95Button>Save</Win95Button>
        <Win95Button>Cancel</Win95Button>
      </div>
    </DialogContent>
  </Dialog>
</AutoPreview>
